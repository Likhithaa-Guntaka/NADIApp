Animation CSS — Paste into Figma Make global styles

Required for pod breathing effect on all wristband SVGs 
@keyframes breathe {
  0%, 100% { transform: scale(1);    opacity: 0.72; }
  50%       { transform: scale(1.13); opacity: 1;    }
}
@keyframes floatLogo {
  0%, 100% { transform: translateY(0px);  }
  50%       { transform: translateY(-8px); }
}

/* Pod breathing — apply to circle elements inside SVG */
.pod-core      { animation: breathe 2.8s ease-in-out infinite;               transform-origin: center; }
.pod-ring      { animation: breathe 2.8s ease-in-out infinite; animation-delay: -1.4s; transform-origin: center; }
.pod-core-fast { animation: breathe 1.8s ease-in-out infinite;               transform-origin: center; }
.pod-ring-fast { animation: breathe 1.8s ease-in-out infinite; animation-delay: -0.9s; transform-origin: center; }
.pod-core-slow { animation: breathe 4s   ease-in-out infinite;               transform-origin: center; }
.pod-ring-slow { animation: breathe 4s   ease-in-out infinite; animation-delay: -2s;   transform-origin: center; }

/* Logo float — apply to SVG wrapper div */
.float-logo    { animation: floatLogo 4s ease-in-out infinite; }
